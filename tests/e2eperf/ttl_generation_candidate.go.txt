package journal

import "sync"

// ttlGeneration owns one generation's integer deadlines. Its caller retains
// Int64SetWithTTL's generation lock; shard locks protect concurrent additions
// and expiry deletions without allocating interface keys or trie nodes.
// No map or value escapes a shard lock. A retired generation is never reused.
type ttlGeneration struct {
	shards [32]ttlDeadlineShard
}

type ttlDeadlineShard struct {
	sync.RWMutex
	deadlines map[int64]int64
	padding   [32]byte // separate adjacent locks on the measured amd64 target
}

func newTTLGeneration() *ttlGeneration { return &ttlGeneration{} }

func (g *ttlGeneration) shard(id int64) *ttlDeadlineShard {
	// Multiplicative hashing spreads sequential and signed IDs across shards.
	return &g.shards[(uint64(id)*0x9e3779b97f4a7c15)>>59]
}

func (g *ttlGeneration) Load(id int64) (int64, bool) {
	s := g.shard(id)
	s.RLock()
	deadline, ok := s.deadlines[id]
	s.RUnlock()
	return deadline, ok
}

func (g *ttlGeneration) Swap(id, deadline int64) (int64, bool) {
	s := g.shard(id)
	s.Lock()
	if s.deadlines == nil {
		s.deadlines = make(map[int64]int64)
	}
	old, loaded := s.deadlines[id]
	s.deadlines[id] = deadline
	s.Unlock()
	return old, loaded
}

func (g *ttlGeneration) Store(id, deadline int64) { g.Swap(id, deadline) }

func (g *ttlGeneration) LoadAndDelete(id int64) (int64, bool) {
	s := g.shard(id)
	s.Lock()
	deadline, ok := s.deadlines[id]
	delete(s.deadlines, id)
	s.Unlock()
	return deadline, ok
}
